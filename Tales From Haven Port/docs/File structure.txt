Tales From Haven Port       # Project Root Directory
    assets                  # All binary, non-code assets
        icons               # Small UI images (item icons, skill icons)
        images              # Larger images (backgrounds, concept art)
        audio               # Sound effects and music
    docs                    # Project Documentation & Design
        design.md            # The core game design document
    src                     # SOURCE CODE: All the game's logic and data (The Most Important Folder)│   │
        data                # GAME DATA: Defines *what* exists in the game world.
            Unique      # Data pools for generating random content.
                Main Questlines
                KeyItems    # Unique, story-critical items 
                KeyLocations    # Major, hand-designed areas
                KeyNpcs     # Important, plot-crucial characters
                KeyOrigins                    
                KeyTraits
                jobTemplates.js  # Templates for creating radiant quests.
                names.js         # Lists of first and last names for NPC generation.
                gangs.js         # generates diverse criminal groups for your game
                npcTraits.js     # Personality and appearance descriptors for NPCs.
                roomTemplates.js # Components for generating building interiors.
            Universal          # Hand-authored, core game content. The "skeleton" of the world.
                attributes.js    # Definitions for STR, DEX, INT, etc.
                weaknesses.js
                events.js        # Definitions for dynamic world events (hurricanes, fires).
                items.js         # Everything from weapons and drugs to health packs.
                keyLocations.js  # Major, hand-designed areas (Neon Lotus Club, Police HQ).
                locations.js     # Definitions for generic locations (e.g., "apartment building").
                mainStory.js     # Flags and nodes for the central narrative.
                npcs.js          # Definitions for generic NPC archetypes ("gang_member").
                paths.js         # Definitions for the playable paths (Pimp, Detective, etc.).
                sexScenes.js     # Definitions for actions, scenes, and descriptors. (NEW)
                skills.js        # Definitions for skills (Seduction, Firearms, Driving).
                vehicles.js      # Definitions for vehicle types and stats. (NEW)
        engine/              # GAME SYSTEMS: The logic for *how* things work. The "brains".
            CombatEngine.js      # Resolves turn-based fights, calculates damage.
            DialogueEngine.js    # Manages conversation trees and NPC interactions.
            EconomyEngine.js     # Handles buying/selling, prices, and the in-game market.
            EventEngine.js       # Processes triggers for dynamic world events.
            GameEngine.js        # The main loop; manages time passage and core state.
            JobGenerator.js      # Creates procedural radiant quests.
            LocationGenerator.js # Generates interiors for generic buildings.
            NPCEquipmentGenerator.js # Assigns gear to NPCs based on their role.
            NPCGenerator.js      # Creates procedural generic NPCs.
            PoliceEngine.js      # Handles crime reporting, chases, and arrests.
            RelationshipEngine.js   #Handles relationships
            SexEngine.js         # Handles consent checks and encounter resolution. (NEW)
            StateManager.js      # Saves and loads the game to/from localStorage.
            VehicleEngine.js     # Handles travel, chases, and vehicle mechanics. (NEW)
        models/              # BLUEPRINTS: JavaScript Classes to create object instances.
            Item.js              # Class for an item instance (with properties like `currentDurability` for a weapon).
            Location.js          # Class for a location instance (e.g., tracks if it's burned down).
            NPC.js               # Class for an NPC instance (inventory, relationship to player).
            PlayerCharacter.js   # The Player's character class (stats, skills, inventory).
            Vehicle.js           # Class for a vehicle instance (current fuel, damage, trunk inventory). (NEW)
        utils/               # UTILITIES: Helper functions used across the project.
            constants.js         # Shared game constants (e.g., `SKILL_NAMES`, `PATH_NAMES`).
            diceRoller.js        # Functions for rolling dice and making skill checks.
            SeededRNG.js         # The seeded random number generator for reproducible randomness.
            textHelpers.js       # Functions for formatting text and generating descriptions.
        components          # UI COMPONENTS: React components that render the game to the screen.
            game                # Components used during gameplay.
                ChaseSequence.js     # UI for the police chase mini-game. (NEW)
                DialogueBox.js       # Displays conversation text and choices.
                CharacterCreator.js
                InventoryPanel.js    # Shows the player's inventory.
                LocationView.js      # Shows the current location's description and exits.
                SexMenu.js           # UI for intimate encounters. (NEW)
                TitleMenu.js
                VehicleMenu.js       # UI for interacting with a car (trunk, refuel). (NEW)
            ui                 # Reusable generic UI components.
                Button.js           # A styled button element.
                Modal.js            # A pop-up dialog box.
                StatsBar.js         # A bar to display health, stamina, etc.
        contexts            # REACT CONTEXTS: For global state management.
            GameContext.js       # Makes the core `gameState` object available anywhere in the UI.
        App.js                  # The main React component that ties everything together.
        index.js                # The JavaScript entry point that renders the `App` component.
    styles                 # STYLESHEETS: CSS files for styling the UI.
        components             # CSS for specific components (e.g., `SexMenu.css`).
        main.css                # Global CSS styles and variables.
    index.html              # The single HTML page that boots the entire application.
